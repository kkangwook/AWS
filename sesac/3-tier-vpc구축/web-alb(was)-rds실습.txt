=> 즉 ALB는 80번으로 받아서 5000번으로 보내는거

from flask import Flask, request
import pymysql
import socket
from werkzeug.serving import WSGIRequestHandler

app = Flask(__name__)
APP_VERSION = "v1"

class RealIPHandler(WSGIRequestHandler): 
    def address_string(self): 
        return self.headers.get("X-Real-IP") or super().address_string()

@app.route("/")
def index(): 
    conn = pymysql.connect( 
        host="RDS_ENDPOINT", 
        user="webuser", 
        password="password", 
        database="testdb" 
    ) 

    cursor = conn.cursor() 
    cursor.execute("SELECT * FROM products") 
    products = cursor.fetchall() 
    cursor.close() 
    conn.close() 

    hostname = socket.gethostname() 
    html = f"<h2>WAS Server: {hostname} / Version: {APP_VERSION}</h2>" 
    html += "<h1>상품 목록</h1><ul>" 

    for product in products: 
        html += f"<li>{product[0]} / {product[1]} / {product[2]}원</li>" 
    html += "</ul>" 
    return html

@app.route("/health")
def health(): 
    return "OK\n"

@app.route("/info")
def info(): 
    return f"WAS={socket.gethostname()} VERSION={APP_VERSION}\n"

@app.route("/stress")
def stress(): 
    load = int(request.args.get("load", 1)) 
    load = max(1, min(load, 5)) 

    result = 0 
    for i in range(load * 100000): 
        result += i * i 

    return (f"WAS={socket.gethostname()} " 
        f"VERSION={APP_VERSION} load={load}\n")

if __name__ == "__main__": 
    app.run(host="0.0.0.0", port=5000, request_handler=RealIPHandler)
